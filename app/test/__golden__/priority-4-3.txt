===== dgmgrl -- DGMGRL -- the broker configuration =====

CONNECT sys@prm01 AS SYSDBA;

CREATE CONFIGURATION 'dg_config' AS PRIMARY DATABASE IS 'prm01' CONNECT IDENTIFIER IS prm01;
ADD DATABASE 'sby01' AS CONNECT IDENTIFIER IS sby01;
ADD DATABASE 'sby02' AS CONNECT IDENTIFIER IS sby02;

EDIT DATABASE 'sby01' SET PROPERTY 'LogXptMode' = 'SYNC';
EDIT DATABASE 'sby02' SET PROPERTY 'LogXptMode' = 'ASYNC';
-- default;
EDIT DATABASE 'prm01' SET PROPERTY 'LogXptMode' = 'SYNC';
-- prm01 ships redo with it once a role swap makes it the standby;

EDIT DATABASE 'prm01' SET PROPERTY 'RedoRoutes' = '(LOCAL : (sby01 SYNC PRIORITY=1, sby02 ASYNC PRIORITY=2))';
-- the routes are pre-declared for both directions: a rule is active when its redo source names the current primary, and nothing rewrites on a role change;
-- PRIORITY is RedoRoutes syntax, GROUP is not -- GROUP is a LOG_ARCHIVE_DEST_n attribute and appears in the init.ora block instead;

EDIT DATABASE 'sby01' SET PROPERTY 'NetTimeout' = 30;
-- default;
EDIT DATABASE 'sby01' SET PROPERTY 'MaxFailure' = 3;
-- the broker default is 1 here -- 1 for a member inside a parenthesised RedoRoutes group, 0 otherwise;
EDIT DATABASE 'sby02' SET PROPERTY 'NetTimeout' = 30;
-- default;
EDIT DATABASE 'sby02' SET PROPERTY 'MaxFailure' = 3;
-- the broker default is 1 here -- 1 for a member inside a parenthesised RedoRoutes group, 0 otherwise;
-- ReopenSecs and MaxFailure are broker properties, not deprecated in any edition, the lines above set them where they differ from the broker default, and the init.ora block shows the REOPEN and MAX_FAILURE attributes they write;

ENABLE CONFIGURATION;

EDIT CONFIGURATION SET PROTECTION MODE AS MAXAVAILABILITY;
-- if the primary does not receive acknowledgement from at least one synchronized standby, then it operates as if it were in maximum performance mode;

EDIT CONFIGURATION SET PROPERTY FastStartFailoverThreshold = 30;
-- default;
-- the clock starts when the observer first loses its connection to the primary, and a failover needs BOTH the observer and the target standby to have lost the primary for that long -- ObserverOverride is FALSE, the broker default, and this page models the default only;
-- ObserverPingInterval and ObserverPingRetry stay unset -- detection runs on the threshold;
-- FastStartFailoverLagLimit is not set -- the 19c broker default is 30 seconds, and in maximum availability a live 19.27 system prints it "(not in use)": the standby must be synchronized instead, which is about redo received, so an apply lag does not block it. The 19c documentation contradicts itself here -- its fast-start failover chapter says zero data loss in maximum availability needs the property set to zero, its property page says the property is used in maximum performance mode;
EDIT CONFIGURATION SET PROPERTY FastStartFailoverPmyShutdown = TRUE;
-- default;
-- TRUE is the default: a primary stalled longer than the threshold, able to reach neither the observer nor the target standby, will shut down;
EDIT CONFIGURATION SET PROPERTY FastStartFailoverAutoReinstate = TRUE;
-- it reinstates only after an observer-initiated failover caused by a primary crash or isolation -- never after a user-configurable health condition or a DBMS_DG.INITIATE_FS_FAILOVER call -- and it needs Flashback Database on both databases, the former primary started and mounted, and the master observer able to reach it;
EDIT DATABASE 'prm01' SET PROPERTY 'FastStartFailoverTarget' = 'sby01';
-- with more than one standby database the property must be set explicitly;
-- the target can only be the name of a physical or logical standby -- never a snapshot standby, a far sync instance or a Zero Data Loss Recovery Appliance -- and it must receive redo directly from the primary database, receiving it from a far sync instance is the sanctioned exception;
-- targets are verified when fast-start failover is enabled, and the reciprocal value is set on the target automatically;

ENABLE FAST_START FAILOVER;

START OBSERVER obs01 IN BACKGROUND CONNECT IDENTIFIER IS prm01 FILE IS /u01/app/oracle/dg/obs01/fsfo.dat LOGFILE IS /u01/app/oracle/dg/obs01/observer.log;
-- issued from a DGMGRL session on dc3-obs01, obs01's own host -- not the session that built the configuration above;
-- IN BACKGROUND is the form the documentation prefers: the prompt returns and the observer process keeps running;
-- CONNECT IDENTIFIER IS uses the credentials stored in the Oracle wallet, if the wallet is not configured, then the command will fail to start the observer;
-- the observer belongs on a host of its own, ideally at a third site or third data center -- it should never be deployed at the primary database server or cluster, or on the standby database server or cluster;

===== initora.prm01 -- LOG_ARCHIVE_DEST_n on prm01 -- where the transport properties land =====

# broker-owned on prm01: LogXptMode, NetTimeout, ReopenSecs
# and MaxFailure all land in these attributes. By hand it
# would read:

log_archive_dest_2 =
   'SERVICE=sby01 SYNC AFFIRM NET_TIMEOUT=30
    REOPEN=300 MAX_FAILURE=3
    GROUP=1 PRIORITY=1
    VALID_FOR=(ONLINE_LOGFILES,PRIMARY_ROLE)
    DB_UNIQUE_NAME=sby01'

log_archive_dest_3 =
   'SERVICE=sby02 ASYNC NOAFFIRM
    REOPEN=300 MAX_FAILURE=3
    GROUP=1 PRIORITY=2
    VALID_FOR=(ONLINE_LOGFILES,PRIMARY_ROLE)
    DB_UNIQUE_NAME=sby02'

# NET_TIMEOUT left out under ASYNC: it bounds the SYNC acknowledgement
# wait, and never affects sessions under ASYNC
# NET_TIMEOUT=30 is the default, spelled out -- seconds LGWR waits
# for the standby's acknowledgement; it is never a failover clock
# MAX_FAILURE=3: after 3 consecutive failed reopens the destination is
# disabled with no further access until it is manually reenabled, and
# REOPEN is then treated as unspecified
# whether a successful reopen resets that counter is not documented
# REOPEN only gates eligibility: transport retries a failed destination at
# log switch time, once the last error plus REOPEN is in the past

# LOG_ARCHIVE_DEST_n takes n = 1..31; 1-10 accept LOCATION or SERVICE,
# 11-31 SERVICE only

# ReopenSecs / MaxFailure are the broker properties behind REOPEN / MAX_FAILURE --
# not deprecated in any edition; the DGMGRL script sets them with EDIT DATABASE
# ... SET PROPERTY where they differ from the broker default
# GROUP= and PRIORITY= are LOG_ARCHIVE_DEST_n attributes, never RedoRoutes syntax; writing each parenthesised RedoRoutes group as a GROUP of its own, with each destination's PRIORITY, is this page's reading of the same intent
# destinations carrying GROUP and PRIORITY re-enter service when they are found accessible -- the one documented exception to the manual-reenable rule
